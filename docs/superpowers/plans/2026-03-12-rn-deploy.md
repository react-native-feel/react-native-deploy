> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Node.js CLI that provisions AWS EC2 Mac instances, builds React Native iOS apps remotely, and submits to TestFlight via the App Store Connect CLI (`asc`).

**Architecture:** Orchestrator CLI that shells out to AWS CLI for infrastructure, connects via SSH (node-ssh) to remote EC2 Mac instances for building, and uses `asc` on the instance for signing and TestFlight submission. Config-driven with EAS-style build profiles.

**Tech Stack:** Node.js, Commander, Inquirer, node-ssh, ora, chalk, AWS CLI, asc CLI

import chalk from 'chalk';
import ora from 'ora';

export class Logger {
  constructor(verbose = false) {
    this.verbose = verbose;
    this._spinner = null;
  }

  info(msg) {
    process.stdout.write(chalk.blue('ℹ') + ` ${msg}\n`);
  }

  success(msg) {
    process.stdout.write(chalk.green('✔') + ` ${msg}\n`);
  }

  warn(msg) {
    process.stdout.write(chalk.yellow('⚠') + ` ${msg}\n`);
  }

  error(msg) {
    process.stderr.write(chalk.red('✖') + ` ${msg}\n`);
  }

  step(current, total, msg) {
    process.stdout.write(chalk.cyan(`[${current}/${total}]`) + ` ${msg}\n`);
  }

  debug(msg) {
    if (this.verbose) {
      process.stdout.write(chalk.gray(`  ${msg}\n`));
    }
  }

  spinner(msg) {
    this._spinner = ora(msg).start();
    return this._spinner;
  }

  stopSpinner(success = true, msg) {
    if (this._spinner) {
      if (success) {
        this._spinner.succeed(msg);
      } else {
        this._spinner.fail(msg);
      }
      this._spinner = null;
    }
  }
}

export const logger = new Logger();
```

import { execFileSync, execFile } from 'node:child_process';

// Builds an argument array for execFileSync('aws', args) — avoids shell injection
export function buildAwsArgs(service, action, options = {}) {
  const args = [service, action];
  for (const [key, value] of Object.entries(options)) {
    if (value === true) {
      args.push(`--${key}`);
    } else if (value !== false && value !== null && value !== undefined) {
      args.push(`--${key}`, String(value));
    }
  }
  args.push('--output', 'json');
  return args;
}

// For tests: reconstructs the command string for assertions
export function buildAwsCommand(service, action, options = {}) {
  return ['aws', ...buildAwsArgs(service, action, options)].join(' ');
}

export function parseAwsJson(output) {
  try {
    return JSON.parse(output.trim());
  } catch {
    throw new Error(`Failed to parse AWS CLI output: ${output.substring(0, 200)}`);
  }
}

export function awsExec(service, action, options = {}) {
  const args = buildAwsArgs(service, action, options);
  try {
    const output = execFileSync('aws', args, { encoding: 'utf-8', timeout: 30000 });
    return parseAwsJson(output);
  } catch (err) {
    throw new Error(`AWS CLI error: ${err.message}`);
  }
}

export function awsExecAsync(service, action, options = {}) {
  const args = buildAwsArgs(service, action, options);
  return new Promise((resolve, reject) => {
    execFile('aws', args, { encoding: 'utf-8', timeout: 60000 }, (err, stdout, stderr) => {
      if (err) {
        reject(new Error(`AWS CLI error: ${stderr || err.message}`));
        return;
      }
      try {
        resolve(parseAwsJson(stdout));
      } catch (parseErr) {
        reject(parseErr);
      }
    });
  });
}
```

const REQUIRED_FIELDS = {
  'aws.region': 'string',
  'aws.instanceType': 'string',
  'aws.keyPairName': 'string',
  'aws.securityGroupName': 'string',
  'aws.nodeVersion': 'string',
  'aws.xcodeVersion': 'string',
  'asc.keyId': 'string',
  'asc.issuerId': 'string',
  'asc.privateKeyPath': 'string',
  'repo.url': 'string',
  'repo.branch': 'string',
};

const VALID_BUILD_TYPES = ['expo', 'classic'];
const VALID_BUILD_CONFIGS = ['Debug', 'Release'];

function getNestedValue(obj, path) {
  return path.split('.').reduce((curr, key) => curr?.[key], obj);
}

export function validateConfig(config) {
  const errors = [];

  for (const [path, type] of Object.entries(REQUIRED_FIELDS)) {
    const value = getNestedValue(config, path);
    if (value === undefined || value === null || value === '') {
      errors.push(`Missing required field: ${path}`);
    } else if (typeof value !== type) {
      errors.push(`${path} must be a ${type}, got ${typeof value}`);
    }
  }

  if (config.build) {
    for (const [profileName, profile] of Object.entries(config.build)) {
      if (profile.buildType && !VALID_BUILD_TYPES.includes(profile.buildType)) {
        errors.push(
          `build.${profileName}.buildType must be one of: ${VALID_BUILD_TYPES.join(', ')}`
        );
      }
      if (profile.buildConfiguration && !VALID_BUILD_CONFIGS.includes(profile.buildConfiguration)) {
        errors.push(
          `build.${profileName}.buildConfiguration must be one of: ${VALID_BUILD_CONFIGS.join(', ')}`
        );
      }
    }
  }

  return errors;
}
```

const MAX_EXTENDS_DEPTH = 5;

export function resolveProfile(profiles, name) {
  if (!profiles[name]) {
    throw new Error(`Profile "${name}" not found in config`);
  }

  const chain = [];
  let current = name;

  while (current) {
    if (chain.includes(current)) {
      throw new Error(`Circular extends detected: ${chain.join(' -> ')} -> ${current}`);
    }
    if (chain.length > MAX_EXTENDS_DEPTH) {
      throw new Error(`Extends depth exceeds maximum of ${MAX_EXTENDS_DEPTH}`);
    }
    chain.push(current);
    const profile = profiles[current];
    current = profile.extends || null;
    if (current && !profiles[current]) {
      throw new Error(`Profile "${current}" not found (extended by "${chain.at(-1)}")`);
    }
  }

  // Merge from base to most specific
  chain.reverse();
  const merged = {};
  for (const profileName of chain) {
    const profile = profiles[profileName];
    for (const [key, value] of Object.entries(profile)) {
      if (key === 'extends') continue;
      if (typeof value === 'object' && value !== null && !Array.isArray(value)) {
        merged[key] = { ...merged[key], ...value };
      } else {
        merged[key] = value;
      }
    }
  }

  return merged;
}
```

import { readFileSync, existsSync } from 'node:fs';
import { join, resolve } from 'node:path';
import { homedir } from 'node:os';
import { validateConfig } from './schema.js';

const CONFIG_FILENAME = 'deploy.config.json';

export function loadConfig(dir = process.cwd()) {
  const configPath = join(resolve(dir), CONFIG_FILENAME);

  if (!existsSync(configPath)) {
    throw new Error(
      `deploy.config.json not found in ${dir}. Run "rn-deploy setup" to create one.`
    );
  }

  const raw = readFileSync(configPath, 'utf-8');
  let config;
  try {
    config = JSON.parse(raw);
  } catch {
    throw new Error(`Invalid JSON in ${configPath}`);
  }

  // Expand ~ in privateKeyPath
  if (config.asc?.privateKeyPath?.startsWith('~')) {
    config.asc.privateKeyPath = config.asc.privateKeyPath.replace('~', homedir());
  }

  return config;
}

export function mergeWithFlags(config, flags) {
  const merged = JSON.parse(JSON.stringify(config));

  if (flags.repo) merged.repo.url = flags.repo;
  if (flags.branch) merged.repo.branch = flags.branch;
  if (flags.region) merged.aws.region = flags.region;
  if (flags.scheme) {
    // Applied to resolved profile later
    merged._flagOverrides = merged._flagOverrides || {};
    merged._flagOverrides.scheme = flags.scheme;
  }
  if (flags.buildType) {
    merged._flagOverrides = merged._flagOverrides || {};
    merged._flagOverrides.buildType = flags.buildType;
  }

  return merged;
}

export function validateLoadedConfig(config) {
  const errors = validateConfig(config);
  if (errors.length > 0) {
    throw new Error(
      `Config validation failed:\n${errors.map(e => `  - ${e}`).join('\n')}`
    );
  }
}

import { execSync } from 'node:child_process';
import { logger } from '../utils/logger.js';

export function buildAllocateCommand(az, instanceType, region) {
  // mac2.metal -> mac2 instance family
  const family = instanceType.split('.')[0];
  return `aws ec2 allocate-hosts --instance-family ${family} --availability-zone ${az} --quantity 1 --region ${region} --output json`;
}

export function buildDescribeCommand(hostId, region) {
  return `aws ec2 describe-hosts --host-ids ${hostId} --region ${region} --output json`;
}

export function buildReleaseCommand(hostId, region) {
  return `aws ec2 release-hosts --host-ids ${hostId} --region ${region} --output json`;
}

async function getAvailableAz(instanceType, region) {
  // List AZs that support the instance type
  const output = execSync(
    `aws ec2 describe-instance-type-offerings --location-type availability-zone --filters "Name=instance-type,Values=${instanceType}" --region ${region} --output json`,
    { encoding: 'utf-8' }
  );
  const data = JSON.parse(output);
  const azs = data.InstanceTypeOfferings.map(o => o.Location);
  if (azs.length === 0) {
    throw new Error(
      `Instance type ${instanceType} is not available in ${region}. Try a different region.`
    );
  }
  return azs[0];
}

export async function allocateDedicatedHost(instanceType, region) {
  const az = await getAvailableAz(instanceType, region);
  logger.info(`Allocating Dedicated Host in ${az}...`);

  const output = execSync(buildAllocateCommand(az, instanceType, region), {
    encoding: 'utf-8',
  });
  const data = JSON.parse(output);
  const hostId = data.HostIds[0];

  logger.info(`Dedicated Host allocated: ${hostId}. Waiting for availability...`);

  // Wait for host to become available
  let attempts = 0;
  const maxAttempts = 60; // 5 minutes at 5s intervals
  while (attempts < maxAttempts) {
    const descOutput = execSync(buildDescribeCommand(hostId, region), {
      encoding: 'utf-8',
      stdio: 'pipe',
    });
    const descData = JSON.parse(descOutput);
    const state = descData.Hosts[0].State;

    if (state === 'available') {
      logger.success(`Dedicated Host ${hostId} is available`);
      return { hostId, az };
    }

    if (state === 'released' || state === 'declined') {
      throw new Error(`Dedicated Host ${hostId} entered unexpected state: ${state}`);
    }

    attempts++;
    await new Promise(resolve => setTimeout(resolve, 5000));
  }

  throw new Error(`Timed out waiting for Dedicated Host ${hostId} to become available`);
}

export async function releaseDedicatedHost(hostId, region) {
  logger.info(`Releasing Dedicated Host ${hostId}...`);
  try {
    execSync(buildReleaseCommand(hostId, region), { stdio: 'pipe' });
    logger.success(`Dedicated Host ${hostId} released`);
  } catch (err) {
    const msg = err.message || '';
    if (msg.includes('within its minimum allocation period')) {
      logger.warn(
        `Cannot release host ${hostId} yet — still within the 24-hour minimum allocation period.`
      );
      return false;
    }
    throw err;
  }
  return true;
}

export async function getHostState(hostId, region) {
  try {
    const output = execSync(buildDescribeCommand(hostId, region), {
      encoding: 'utf-8',
      stdio: 'pipe',
    });
    const data = JSON.parse(output);
    return data.Hosts[0];
  } catch {
    return null;
  }
}

import { logger } from '../utils/logger.js';

export function buildRunCommand({ amiId, instanceType, keyName, sgId, hostId, region }) {
  return `aws ec2 run-instances --image-id ${amiId} --instance-type ${instanceType} --key-name ${keyName} --security-group-ids ${sgId} --placement "HostId=${hostId}" --region ${region} --output json`;

  return `aws ec2 terminate-instances --instance-ids ${instanceId} --region ${region} --output json`;
}

async function getLatestMacAmi(region) {
  const output = execSync(
    `aws ec2 describe-images --owners amazon --filters "Name=name,Values=amzn-ec2-macos-*" "Name=architecture,Values=arm64_mac" --query "Images | sort_by(@, &CreationDate) | [-1]" --region ${region} --output json`,
    { encoding: 'utf-8' }
  );
  const ami = JSON.parse(output);
  if (!ami || !ami.ImageId) {
    throw new Error(`No macOS AMI found in ${region}`);
  }
  logger.debug(`Using AMI: ${ami.ImageId} (${ami.Name})`);
  return ami.ImageId;
}

export async function launchInstance({ instanceType, keyName, sgId, hostId, region, customAmiId }) {
  const amiId = customAmiId || await getLatestMacAmi(region);

  logger.info('Launching EC2 Mac instance...');
  const output = execSync(
    buildRunCommand({ amiId, instanceType, keyName, sgId, hostId, region }),
    { encoding: 'utf-8' }
  );
  const data = JSON.parse(output);
  const instanceId = data.Instances[0].InstanceId;

  logger.info(`Instance ${instanceId} launched. Waiting for it to be running...`);

  // Wait for instance to be running
  execSync(
    `aws ec2 wait instance-running --instance-ids ${instanceId} --region ${region}`,
    { encoding: 'utf-8', timeout: 600000 }
  );

  // Wait for status checks
  logger.info('Waiting for status checks to pass...');
  execSync(
    `aws ec2 wait instance-status-ok --instance-ids ${instanceId} --region ${region}`,
    { encoding: 'utf-8', timeout: 600000 }
  );

  // Get public IP
  const descOutput = execSync(
    `aws ec2 describe-instances --instance-ids ${instanceId} --region ${region} --output json`,
    { encoding: 'utf-8' }
  );
  const descData = JSON.parse(descOutput);
  const publicIp = descData.Reservations[0].Instances[0].PublicIpAddress;

  if (!publicIp) {
    throw new Error(`Instance ${instanceId} has no public IP. Check your VPC/subnet settings.`);
  }

  logger.success(`Instance ready: ${instanceId} at ${publicIp}`);
  return { instanceId, publicIp, amiId };
}

      `aws ec2 describe-images --image-ids ${amiId} --region ${region} --output json`,

    );
    const data = JSON.parse(output);

      ?.filter(b => b.Ebs?.SnapshotId)

import { NodeSSH } from 'node-ssh';
import { logger } from '../utils/logger.js';

export function buildSshConfig(host, privateKeyPath, username = 'ec2-user') {
  return {
    host,
    username,
    privateKeyPath,
    readyTimeout: 30000,
    keepaliveInterval: 10000,
  };
}

export async function createSshConnection(host, privateKeyPath, retries = 3) {
  const config = buildSshConfig(host, privateKeyPath);
  const ssh = new NodeSSH();

  for (let attempt = 1; attempt <= retries; attempt++) {
    try {
      await ssh.connect(config);
      logger.success(`SSH connected to ${host}`);
      return ssh;
    } catch (err) {
      if (attempt === retries) {
        throw new Error(`SSH connection failed after ${retries} attempts: ${err.message}`);
      }
      const delay = Math.pow(2, attempt) * 1000;
      logger.warn(`SSH attempt ${attempt}/${retries} failed. Retrying in ${delay / 1000}s...`);
      await new Promise(r => setTimeout(r, delay));
    }
  }
}

export async function sshExec(ssh, command, { stream = false, cwd, env } = {}) {
  logger.debug(`> ${command}`);

  if (stream) {
    return new Promise((resolve, reject) => {
      ssh.execCommand(command, {
        cwd,
        env,
        onStdout: (chunk) => process.stdout.write(chunk),
        onStderr: (chunk) => process.stderr.write(chunk),
      }).then(result => {
        if (result.code !== 0) {
          reject(new Error(`Command failed (exit ${result.code}): ${command}`));
        } else {
          resolve(result);
        }
      }).catch(reject);
    });
  }

  const result = await ssh.execCommand(command, { cwd, env });
  if (result.code !== 0) {
    throw new Error(`Command failed (exit ${result.code}): ${result.stderr || result.stdout}`);
  }
  return result;
}

export async function sshUpload(ssh, localPath, remotePath) {
  logger.debug(`Uploading ${localPath} -> ${remotePath}`);
  await ssh.putFile(localPath, remotePath);
}

export async function sshDownload(ssh, remotePath, localPath) {
  logger.debug(`Downloading ${remotePath} -> ${localPath}`);
  await ssh.getFile(localPath, remotePath);
}

    clearState();

  // Cost warning
  if (!options.nonInteractive) {

      logger.info('Cancelled');
      return null;
    }
  }

  // Provision
  const keyPath = await ensureKeyPair(keyPairName, region);
  const sgId = await ensureSecurityGroup(securityGroupName, region);
  const { hostId, az } = await allocateDedicatedHost(instanceType, region);
  const { instanceId, publicIp, amiId } = await launchInstance({
    instanceType,

    sgId,
    hostId,
    region,
    customAmiId,
  });

    instanceId,
    publicIp,
    allocatedAt: new Date().toISOString(),
    region,
    amiId,
    keyPath,
    sgId,
  };

export async function infraDown(options = {}) {

    logger.info('No active infrastructure found');
    return;

  await terminateInstance(state.instanceId, region);

import { loadConfig, mergeWithFlags, validateLoadedConfig } from '../config/loader.js';
import { resolveProfile } from '../config/profiles.js';
import { loadState } from '../infra/state.js';
import { infraUp, infraDown } from './infra.js';
import { createSshConnection, sshExec } from '../remote/ssh.js';
import { provisionInstance, cloneRepo } from '../remote/provision.js';
import { setupSigning, fetchSigningAssets, generateExportOptionsPlist } from '../remote/signing.js';
import { getExpoBuildSteps } from '../build/expo.js';
import { getClassicBuildSteps } from '../build/classic.js';
import { findAndDownloadIpa } from '../build/artifacts.js';
import { createCustomAmi } from '../infra/ami.js';
import { getPrivateKeyPath } from '../infra/key-pair.js';
import { logger } from '../utils/logger.js';
import { confirm, input, select } from '../utils/prompts.js';
import { verifyPrerequisites } from '../utils/prerequisites.js';
import { submitToTestFlight } from './submit.js';

const TMUX_SESSION = 'rn-deploy-build';
const TMUX_LOG = '/tmp/rn-deploy-build.log';

export async function buildCommand(options = {}) {
  await verifyPrerequisites();

  // Handle --resume: reconnect to existing tmux build session
  if (options.resume) {
    return resumeBuild(options);
  }

  // Load and merge config
  const rawConfig = loadConfig();
  const config = mergeWithFlags(rawConfig, options);
  validateLoadedConfig(config);

  // Resolve build profile
  const profileName = options.profile || 'production';
  let profile = resolveProfile(config.build, profileName);

  // Apply flag overrides
  if (config._flagOverrides) {
    profile = { ...profile, ...config._flagOverrides };
  }

  // Interactive prompts for missing values
  if (!options.nonInteractive) {
    if (!config.repo.url) {
      config.repo.url = await input('Git repository URL:');
    }
    if (!profile.scheme) {
      profile.scheme = await input('Xcode scheme name:');
    }
    if (!profile.buildType) {
      profile.buildType = await select('Build type:', ['expo', 'classic']);
    }
  }

  const { region, keyPairName } = config.aws;
  const totalTimeoutMs = (options.timeout || 60) * 60 * 1000;
  const buildStartTime = Date.now();

  // Step 1: Ensure infrastructure
  logger.step(1, 6, 'Provisioning infrastructure');
  let state = loadState();
  if (!state || !(await isInstanceReachable(state))) {
    state = await infraUp(options);
    if (!state) return;
  } else {
    logger.info(`Reusing existing instance: ${state.instanceId} at ${state.publicIp}`);
  }

  // Step 2: SSH connect
  logger.step(2, 6, 'Connecting via SSH');
  const keyPath = state.keyPath || getPrivateKeyPath(keyPairName);
  const ssh = await createSshConnection(state.publicIp, keyPath);

  try {
    // Step 3: Provision instance (skip if custom AMI)
    if (!config.aws.customAmiId) {
      logger.step(3, 6, 'Setting up build environment');
      await provisionInstance(ssh, {
        nodeVersion: config.aws.nodeVersion,
        xcodeVersion: config.aws.xcodeVersion,
      });
    } else {
      logger.step(3, 6, 'Using custom AMI — skipping provisioning');
    }

    // Setup signing and extract real values
    await setupSigning(ssh, config.asc);
    const signingInfo = await fetchSigningAssets(ssh);

    // Clone repo
    const workDir = await cloneRepo(ssh, config.repo.url, config.repo.branch);

    // Generate ExportOptions.plist on remote using real signing values
    const distribution = profile.distribution || 'app-store';
    const method = distribution === 'ad-hoc' ? 'ad-hoc' : 'app-store';
    const plistContent = generateExportOptionsPlist({
      method,
      teamId: signingInfo.teamId,
      bundleId: signingInfo.bundleId,
      profileName: signingInfo.profileName,
    });
    await sshExec(ssh, `cat > ${workDir}/ExportOptions.plist << 'PLIST_EOF'\n${plistContent}\nPLIST_EOF`);

    // Step 4: Build inside tmux session for resilience
    logger.step(4, 6, 'Building app');
    const buildSteps = profile.buildType === 'expo'
      ? getExpoBuildSteps({ scheme: profile.scheme, buildConfiguration: profile.buildConfiguration, workDir })
      : getClassicBuildSteps({ scheme: profile.scheme, buildConfiguration: profile.buildConfiguration, workDir });

    // Build env vars as export string for shell persistence
    const envExports = Object.entries(profile.env || {})
      .map(([k, v]) => `export ${k}="${v}"`)
      .join(' && ');

    // Create a build script with all steps, run inside tmux
    const buildScript = buildSteps.map(s => `echo "=== ${s.name} ===" && ${s.cmd}`).join(' && ');
    const fullScript = envExports ? `${envExports} && ${buildScript}` : buildScript;

    await sshExec(ssh, `tmux kill-session -t ${TMUX_SESSION} 2>/dev/null || true`);
    await sshExec(ssh, `tmux new-session -d -s ${TMUX_SESSION} 'bash -c "${fullScript.replace(/"/g, '\\"')}" 2>&1 | tee ${TMUX_LOG}'`);

    // Tail the log and wait for tmux session to finish
    logger.info('Build started in tmux session. Streaming logs...');
    const tailPromise = sshExec(ssh, `tail -f ${TMUX_LOG} &
      while tmux has-session -t ${TMUX_SESSION} 2>/dev/null; do sleep 5; done
      kill %1 2>/dev/null
      # Check if build succeeded (last command exit code)
      tail -1 ${TMUX_LOG}`, { stream: true });

    const timeoutPromise = new Promise((_, reject) => {
      const remaining = totalTimeoutMs - (Date.now() - buildStartTime);
      setTimeout(() => reject(new Error(`Build timed out after ${options.timeout || 60} minutes`)), remaining);
    });

    await Promise.race([tailPromise, timeoutPromise]);

    // Step 5: Retrieve artifact
    logger.step(5, 6, 'Retrieving build artifact');
    const exportPath = `${workDir}/build/export`;
    const { localPath, remotePath } = await findAndDownloadIpa(
      ssh, exportPath, process.cwd(), profile.scheme
    );

    // Step 6: Submit (if auto-submit)
    if (options.autoSubmit) {
      logger.step(6, 6, 'Submitting to TestFlight');
      await submitToTestFlight(ssh, {
        ipaPath: remotePath,
        config,
        profileName,
        whatToTest: options.message, // --message flag used as "What to Test"
      });
    } else {
      logger.step(6, 6, 'Build complete');
      logger.success(`IPA saved to: ${localPath}`);
    }

    // Offer AMI caching after first build with no custom AMI
    if (!config.aws.customAmiId && !options.nonInteractive) {
      const saveAmi = await confirm('Save this instance as a custom AMI for faster future builds?', true);
      if (saveAmi) {
        const amiId = await createCustomAmi(state.instanceId, region);
        logger.info(`Add this to your deploy.config.json aws section: "customAmiId": "${amiId}"`);
      }
    }
  } finally {
    ssh.dispose();
  }

  // Teardown unless --no-teardown
  if (!options.noTeardown) {
    await infraDown(options);
  } else {
    logger.info('Instance kept alive (--no-teardown). Run "rn-deploy infra down" when done.');
  }
}

async function resumeBuild(options) {
  const state = loadState();
  if (!state) {
    throw new Error('No active instance found. Cannot resume.');
  }

  const config = loadConfig();
  const keyPath = state.keyPath || getPrivateKeyPath(config.aws.keyPairName);
  const ssh = await createSshConnection(state.publicIp, keyPath);

  try {
    // Check if tmux session exists
    const check = await sshExec(ssh, `tmux has-session -t ${TMUX_SESSION} 2>/dev/null && echo "running" || echo "done"`);
    if (check.stdout.trim() === 'running') {
      logger.info('Reconnecting to active build session...');
      await sshExec(ssh, `tail -f ${TMUX_LOG} &
        while tmux has-session -t ${TMUX_SESSION} 2>/dev/null; do sleep 5; done
        kill %1 2>/dev/null`, { stream: true });
    } else {
      logger.info('Build session completed. Checking for artifacts...');
      await sshExec(ssh, `cat ${TMUX_LOG}`, { stream: true });
    }
  } finally {
    ssh.dispose();
  }
}

async function isInstanceReachable(state) {
  if (!state || !state.keyPath) return false;
  try {
    // Note: StrictHostKeyChecking=no is acceptable here because EC2 Mac instances
    // are ephemeral — host keys change on every new instance. We verify identity
    // via AWS instance ID + the SSH key pair we created.
    const { execFileSync } = await import('node:child_process');
    execFileSync('ssh', [
      '-o', 'ConnectTimeout=5',
      '-o', 'StrictHostKeyChecking=accept-new',
      '-i', state.keyPath,
      `ec2-user@${state.publicIp}`,
      'echo ok',
    ], { stdio: 'pipe', timeout: 10000 });
    return true;
  } catch {
    return false;
  }
}

    const ipaPath = options.ipa;
    let remoteIpaPath = ipaPath;

    // If local path, upload to instance
    if (ipaPath && !ipaPath.startsWith('/Users/ec2') && !ipaPath.startsWith('~/build-workspace')) {
      remoteIpaPath = '/tmp/upload.ipa';
      logger.info('Uploading .ipa to instance...');
      await sshUpload(ssh, ipaPath, remoteIpaPath);

      remoteIpaPath = await input('Path to .ipa file on instance:');
    }

    await submitToTestFlight(ssh, {
      ipaPath: remoteIpaPath,
      config,
      profileName: options.profile || 'production',
      groups: options.groups,
      whatToTest: options.whatToTest || options.message,

    ssh.dispose();

export async function submitToTestFlight(ssh, { ipaPath, config, profileName, groups, whatToTest }) {
  // Resolve submit profile
  const submitProfile = config.submit?.[profileName] || {};
  const testFlightGroups = groups?.split(',') || submitProfile.testFlightGroups || [];
  const testNotes = whatToTest || submitProfile.whatToTest || '';

  // Get app ID from asc
  logger.info('Looking up app information...');
  const appsResult = await sshExec(ssh, 'asc apps list --json');
  const apps = JSON.parse(appsResult.stdout);
  if (!apps || apps.length === 0) {
    throw new Error('No apps found in App Store Connect. Create one first.');
  }
  const appId = apps[0].id; // TODO: let user select if multiple apps

  // Upload build
  logger.info('Uploading build to App Store Connect...');
  await sshExec(ssh, `asc builds upload --app "${appId}" --ipa "${ipaPath}"`, { stream: true });

  // Distribute to TestFlight groups
  for (const group of testFlightGroups) {
    logger.info(`Adding to TestFlight group: ${group}`);
    await sshExec(ssh, `asc publish testflight --group "${group}" --wait`, { stream: true });
  }

  if (testNotes) {
    logger.info(`Setting "What to Test": ${testNotes}`);
    // Note: asc may not have a direct command for this — would need API call
  }
